// docs/design-doc/workflow-system/节点定义协议.md
interface NodeDefinition {
  type: string; // 唯一标识
  properties: Schema; // 配置 Schema
  execute: (ctx: Context) => Promise<NodeResult>;
}
// 严格限制：禁止在 execute 中进行 DOM 操作
